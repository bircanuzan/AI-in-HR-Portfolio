-- AI in HR Analytics Lab | Example Power BI DAX Measures

Headcount =
DISTINCTCOUNT(HR[employee_id])

Exits 12M =
SUM(HR[exit_12m])

Exit Rate % =
DIVIDE([Exits 12M], [Headcount], 0)

Internal Moves 12M =
SUM(HR[internal_move_12m])

Internal Mobility Rate % =
DIVIDE([Internal Moves 12M], [Headcount], 0)

Average Time to Fill =
AVERAGE(HR[time_to_fill_days])

Average Learning Hours =
AVERAGE(HR[learning_hours])

Average Skill Level =
AVERAGE(HR[skill_level])

Employees Below Proficiency =
CALCULATE(
    [Headcount],
    HR[skill_level] < 3
)
